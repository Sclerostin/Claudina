Get it running

About "Script could not be translated from: |B|W|E|". TradingView shows this when it can't read the script's first line, //@version=6, or when its translator chokes on an unusual construct. DT Core 3 is rewritten in plain ASCII text using only long-established Pine functions. Paste it into an empty editor so nothing old is left above line 1.

  1. Copy DT Core 3 with the button below.
  2. Empty the editor completely. Pine Editor → open DT Core → Select All → Delete. Scroll up and down: the editor must be blank.
  3. Paste and check line 1. It must read exactly //@version=6, with nothing before it on the line. Then Save.
  4. Re-add it to the chart. DT Core is now a strategy, so remove the old copy from the chart (tap its name → remove), then ƒx Indicators → Personal → DT Core.
  5. 5-minute candles, then your numbers. Tap DT Core → Settings → Inputs → Risk: account size and risk per trade. Under Properties, set Initial capital to the same account size.

//@version=6
// DT Core 3 - opening range breakout with automatic BUY / SELL, stop, target and exit
//
// What it does for you (Crabel 1990 opening range breakout, as taught by Aziz 2015):
//   1. Waits for the opening range: the high and low of the first minutes after the open.
//      Auto picks 5, 15 or 30 minutes from your chart timeframe.
//   2. BUY when a candle closes above the range, above VWAP, on strong volume.
//      SELL (short) when a candle closes below the range, below VWAP, on strong volume.
//   3. Stop at the middle of the range. Target at 2x the risk.
//   4. Exit at the stop, the target, or 5 minutes before the close.
//   At most one BUY and one SELL per day, and none after the cutoff.
//
// This is a strategy: TradingView simulates every trade (see Strategy Tester), and
// alerts on order fills can send each order to a broker through a webhook.
// Educational tool. Not financial advice.
strategy("DT Core", overlay = true, initial_capital = 25000, margin_long = 25, margin_short = 25, process_orders_on_close = true, pyramiding = 0, slippage = 1, max_labels_count = 500)

type VolHist
    array<float> vals

var map<int, VolHist> volByTime = map.new<int, VolHist>()
var float cumVol = 0.0

    cumVol := 0.0
cumVol += vol

int   todKey = hour * 60 + minute
float avgCum = na
if rvolOn and map.contains(volByTime, todKey)
    VolHist h = map.get(volByTime, todKey)
    if array.size(h.vals) > 0
        avgCum := array.avg(h.vals)
float rvol = not na(avgCum) and avgCum > 0 ? cumVol / avgCum : na
if rvolOn and barstate.isconfirmed
    if not map.contains(volByTime, todKey)
        map.put(volByTime, todKey, VolHist.new(array.new<float>()))
    VolHist hh = map.get(volByTime, todKey)

// ----- Helpers -----
float  riskUsd = acct * riskPct / 100
float  pv      = syminfo.pointvalue
bool   fracQty = syminfo.type == "crypto" or syminfo.type == "forex"
string unit    = syminfo.type == "futures" ? " contracts" : fracQty ? " units" : " shares"

px(float p) =>
    str.tostring(p, format.mintick)

// Size so the stop loses your risk amount, capped at 4x day-trading buying power
qtyFor(float risk) =>
    float raw = risk > 0 ? riskUsd / (risk * pv) : na
    float cap = close > 0 ? acct * 4 / (close * pv) : na
    float q   = na(cap) ? raw : math.min(raw, cap)
    fracQty ? q : math.max(1, math.floor(q))

qtyStr(float q) =>
    na(q) ? "-" : fracQty ? str.tostring(q, "0.####") : str.tostring(q, "0")

clock(int t) =>
    int hh24 = hour(t, showTz)
    int mm   = minute(t, showTz)
    int h12  = hh24 % 12 == 0 ? 12 : hh24 % 12
    str.tostring(h12) + ":" + (mm < 10 ? "0" : "") + str.tostring(mm) + (hh24 < 12 ? " AM" : " PM")

// Webhook messages in the common TradersPost-style format
jsonEntry(string act, float q, float tgt, float stp) =>
    '{"ticker":"' + syminfo.ticker + '","action":"' + act + '","quantity":' + qtyStr(q) + ',"takeProfit":{"limitPrice":' + px(tgt) + '},"stopLoss":{"type":"stop","stopPrice":' + px(stp) + '}}'

jsonExit() =>
    '{"ticker":"' + syminfo.ticker + '","action":"exit"}'

var int    nDays      = 0

    sellUsed := false
    nDays    += 1

// A trade closed since the last bar: label it and keep score
if strategy.closedtrades > seenClosed
    int   k  = strategy.closedtrades - 1
    float ep = strategy.closedtrades.entry_price(k)
    float xp = strategy.closedtrades.exit_price(k)
    int   d  = strategy.closedtrades.size(k) > 0 ? 1 : -1
    float r  = riskPx > 0 ? (xp - ep) * d / riskPx : 0.0
    bool  hitT = d == 1 ? xp >= tgtPx - 3 * syminfo.mintick : xp <= tgtPx + 3 * syminfo.mintick
    bool  hitS = d == 1 ? xp <= stopPx + 3 * syminfo.mintick : xp >= stopPx - 3 * syminfo.mintick
    string head = hitT ? "TARGET" : hitS ? "STOP" : "EXIT"
    string rTxt = (r >= 0 ? "+" : "") + str.tostring(r, "0.0") + "R"
    sumR += r

    lastResult := (d == 1 ? "BUY " : "SELL ") + head + " " + rTxt
    msg := "EXIT " + syminfo.ticker + " @ " + px(xp) + " - " + head + " " + rTxt
    seenClosed := strategy.closedtrades
    dirNow := 0
    exitedNow := true

// New entries, filled at the close of the signal candle
bool flat     = strategy.position_size == 0 and dirNow == 0
bool canTrade = intra and orReady and beforeCut and not eodBar and flat and not exitedNow
bool vwapUpOK = not useVwapF or na(vwapVal) or close > vwapVal
bool vwapDnOK = not useVwapF or na(vwapVal) or close < vwapVal
bool buySig   = canTrade and not buyUsed and xUp and volOK and vwapUpOK
bool sellSig  = canTrade and allowSell and not sellUsed and xDn and volOK and vwapDnOK

    stopPx  := stp
    tgtPx   := tgt
    riskPx  := rsk

        strategy.entry("BUY", strategy.long, qty = q, alert_message = jsonEntry("buy", q, tgt, stp))

Three ways to execute

Pick the level you're comfortable with. Each one builds on the last.

1
Tap to trade on the chart

For a broker connected to TradingView, or TradingView's Paper Trading account. When BUY prints, tap the chart's Buy button. Copy the share count, take-profit and stop-loss from the status panel's Plan and Size rows into the order ticket, and confirm. About 15 seconds.

2
Get pinged, then one bracket order

Any broker. Add one alert per stock (below) and your phone shows "BUY NVDA @ 102.00 - stop 100.75 - target 104.50 - 100 shares". Place it as a bracket (OCO) order in your broker's app. The exit then runs without you.

3
Fully automatic

Your paid TradingView plan includes webhooks. Each simulated fill on DT Core sends a ready-made order with its stop and target to a bridge service such as TradersPost, which places it at your broker. You don't touch anything.

Alerts for levels 1 and 2: plain-language pings

  1. Open the stock's 5-minute chart. DT Core must be on it.
  2. Add an alert. Tap the alarm-clock button, or long-press the chart → Add alert.
  3. Condition: DT Core → alert() function calls only.
  4. Notify: app push and sound on. Create.
BUY NVDA @ 102.00 - stop 100.75 - target 104.50 - 100 shares (risk $125) EXIT NVDA @ 104.50 - TARGET +2.0R

Level 3 setup: fully automatic

  1. Open a bridge account and connect a paper broker first. TradersPost connects to Alpaca, TradeStation, Tradier, Interactive Brokers, Webull, tastytrade and others, and lists Robinhood for stocks. Check its page for your broker's current limits.
  2. Create a strategy in the bridge and copy its webhook URL. Turn on "use signal quantity" and bracket orders if your broker supports them.
  3. In TradingView, add an alert on the stock's 5-minute chart: Condition DT Core → Order fills only.
  4. Message: delete the default text and enter exactly {{strategy.order.alert_message}}
  5. Notifications: turn on Webhook URL and paste the bridge's URL. Create. Repeat for each stock you trade.
  6. Watch a week of paper fills in the bridge's log before switching it to a live account.

What the bridge receives for each order:

{"ticker":"NVDA","action":"buy","quantity":100,"takeProfit":{"limitPrice":104.50},"stopLoss":{"type":"stop","stopPrice":100.75}} {"ticker":"NVDA","action":"exit"}

  • Settings freezeAn alert keeps the settings it was made with. Change a setting, delete and recreate the alert.
  • SELL means shortConfigure the bridge for long and short, or turn off "Take SELL signals" if your account can't short.
  • Exits are doubled upThe bracket at your broker exits on its own. The extra "exit" message closes anything left open, such as at the end of the day.

Reading the signals

Labels appear when a candle closes and never move or disappear afterward.

  • BUYA candle closed above the opening range, above VWAP, on strong volume. Entry is that candle's closing price.
  • SELLThe mirror image below the range. SELL means sell short. If you only buy stocks, turn it off in settings.
  • Stop. Where the trade is wrong.
  • Target. Twice as far from entry as the stop, so a win pays 2× what a loss costs.
  • Entry price.
  • TARGET +2.0RTarget hit. R means "times your risk": +2R on $125 risk is +$250.
  • STOP -1.0RStop hit: a planned, limited loss.
  • EXITNeither was hit, so the trade closed 5 minutes before the market close.
  • VWAP, the day's volume-weighted average price. Green when price is above it, red when below.
  • Opening range. Breakouts from this box are the signals.
  • Yesterday's high and low.
  • Pre-market high and low. Tap ETH at the bottom right of the chart to show extended hours; futures show the overnight range automatically.
  • Midday, 2 to 4 hours after the open. Breakouts fail more often here.

TradingView also draws its own small order arrows for strategies. To hide them, open DT Core's settings → Style and turn off the trades-on-chart option.

What it does, in order

The rules behind the signals

  1. Mark the opening range. The high and low of the first minutes after the open. Toby Crabel showed in 1990 that breakouts from it carry follow-through; Andrew Aziz teaches the same trade for stocks.
  2. Wait for a candle to close outside it. With strong volume, and on the side of VWAP that agrees (Shannon: VWAP is the day's fair price).
  3. Size from the stop. Stop at the middle of the range; shares = dollars at risk ÷ distance to the stop (Tharp).
  4. Let the trade end itself. Target at 2× risk, stop, or 5 minutes before the close. At most one BUY and one SELL a day, none after the cutoff.
  • 0.5–1% per tradeElder's ceiling is 2%. Start small while you learn.
  • Daily stop at −3RThree full losses and you're done for the day.
  • Paper trade firstRun 20 or more signals on TradingView's Paper Trading account before using real money. Many days produce no signal; that's by design.
  • Check your broker's rulesUS margin accounts face pattern-day-trader requirements.

Earlier questions

I saved the script. Why is nothing on the chart?

Saving and adding are two separate steps, and you need both. Saving stores the code in your TradingView account. If it saved without a red error message, the code compiled fine. Adding puts it on a chart:

On the chart, tap ƒx Indicators → Personal (some versions say My scripts) → tap DT Core. Its name appears in the top-left corner of the chart.

Added but still blank?

  • Daily chart. DT Core only draws on intraday charts. Switch to 5 minutes; the panel says so if you forget.
  • Indicator limit. The free plan fits very few indicators per chart. Remove Volume or anything else you're not using (tap its name → remove).
  • Hidden. Tap DT Core's name at the top left and make sure the eye icon is on.
  • Before the open or on a weekend. You'll see earlier days' lines and signals; today's start at 9:30.

What do "0930-1000" and "0930-0935" mean?

They were time windows on a 24-hour clock in New York time, written start-end without colons. DT Core no longer asks for them: you pick Opening range length: Auto, 5, 15 or 30 minutes, and it measures from the opening bell on its own.

The chart settings didn't match my app.

Menus differ between app versions, and most of those settings didn't affect the signals. DT Core handles the timezone inside the script. The only chart settings you need: Candles, the 5-minute timeframe, and ETH on if you want pre-market lines.

DT Momentum · squeeze, 3/10 or RSI in one pane

Pick one oscillator under Settings → Oscillator. Squeeze (Carter): red dots mean energy is building; the first green dot is the release. 3/10 (Raschke) and RSI (Wilder) are the other choices.

DT_Momentum.pine · 95 lines

//@version=6
// ======================================================================
//  DT Momentum - three classic day-trading oscillators in ONE pane
//  Squeeze  - John Carter, Mastering the Trade (Bollinger inside Keltner)
//  3/10     - Linda Bradford Raschke's 3/10 oscillator (SMA3 - SMA10, 16 signal)
//  RSI      - J. Welles Wilder, New Concepts in Technical Trading Systems
//  Pick one in settings; alerts follow the selected oscillator.
//  Educational tool. Not financial advice.
// ======================================================================
indicator("DT Momentum", overlay = false, precision = 2)

string mode = input.string("Squeeze", "Oscillator", options = ["Squeeze", "3/10", "RSI"])

string G_SQ = "Squeeze (Carter)"
int   sqLen  = input.int(20, "Length", minval = 2, group = G_SQ)
float bbMult = input.float(2.0, "Bollinger std dev", minval = 0.1, step = 0.1, group = G_SQ)
float kcMult = input.float(1.5, "Keltner ATR multiple", minval = 0.1, step = 0.1, group = G_SQ)

string G_310 = "3/10 oscillator (Raschke)"
int fastLen = input.int(3, "Fast SMA", minval = 1, group = G_310)
int slowLen = input.int(10, "Slow SMA", minval = 2, group = G_310)
int sigLen  = input.int(16, "Signal SMA", minval = 1, group = G_310)

string G_RSI = "RSI (Wilder)"
int   rsiLen = input.int(14, "Length", minval = 2, group = G_RSI)
float rsiHi  = input.float(70, "Overbought", minval = 50, maxval = 100, group = G_RSI)
float rsiLo  = input.float(30, "Oversold", minval = 0, maxval = 50, group = G_RSI)

string G_CL = "Colors"
color cUp   = input.color(#26a69a, "Up", inline = "c1", group = G_CL)
color cDn   = input.color(#ef5350, "Down", inline = "c1", group = G_CL)
color cSqz  = input.color(#ef5350, "Squeeze on", inline = "c2", group = G_CL)
color cFree = input.color(#26a69a, "Squeeze off", inline = "c2", group = G_CL)
color cLine = input.color(#42a5f5, "Line", inline = "c3", group = G_CL)
color cSig  = input.color(#f5a623, "Signal", inline = "c3", group = G_CL)

bool isSq  = mode == "Squeeze"
bool is310 = mode == "3/10"
bool isRsi = mode == "RSI"

// ----- Squeeze: Bollinger Bands inside Keltner Channels = volatility coiled
float basis = ta.sma(close, sqLen)
float bbDev = bbMult * ta.stdev(close, sqLen)
float kcDev = kcMult * ta.atr(sqLen)
bool  sqzOn = bbDev < kcDev
bool  sqzFired = sqzOn[1] and not sqzOn
float mom = ta.linreg(close - math.avg(math.avg(ta.highest(high, sqLen), ta.lowest(low, sqLen)), basis), sqLen, 0)
float momPrev = nz(mom[1])
color momCol = mom >= 0 ? (mom >= momPrev ? cUp : color.new(cUp, 60)) : (mom <= momPrev ? cDn : color.new(cDn, 60))

plot(isSq ? mom : na, "Squeeze momentum", momCol, 4, plot.style_columns)
plot(isSq ? 0.0 : na, "Squeeze state", sqzOn ? cSqz : cFree, 3, plot.style_circles)

// ----- 3/10 oscillator
float osc    = ta.sma(close, fastLen) - ta.sma(close, slowLen)
float oscSig = ta.sma(osc, sigLen)
plot(is310 ? osc - oscSig : na, "3/10 histogram", osc >= oscSig ? color.new(cUp, 70) : color.new(cDn, 70), 4, plot.style_columns)
plot(is310 ? osc : na, "3/10 fast line", cLine, 2)
plot(is310 ? oscSig : na, "3/10 slow line", cSig, 2)

// ----- RSI
float rsi = ta.rsi(close, rsiLen)
plot(isRsi ? rsi : na, "RSI", cLine, 2)
hline(rsiHi, "Overbought", color.new(cDn, 40), hline.style_dashed, display = isRsi ? display.all : display.none)
hline(50, "Midline", color.new(color.gray, 50), hline.style_dotted, display = isRsi ? display.all : display.none)
hline(rsiLo, "Oversold", color.new(cUp, 40), hline.style_dashed, display = isRsi ? display.all : display.none)
hline(0, "Zero", color.new(color.gray, 60), hline.style_dotted, display = isRsi ? display.none : display.all)

// ----- Signals
bool x310Up = ta.crossover(osc, oscSig)
bool x310Dn = ta.crossunder(osc, oscSig)
bool xRsiUp = ta.crossover(rsi, rsiLo)
bool xRsiDn = ta.crossunder(rsi, rsiHi)

bool sqUp  = isSq and sqzFired and mom > 0
bool sqDn  = isSq and sqzFired and mom < 0
bool t3Up  = is310 and x310Up
bool t3Dn  = is310 and x310Dn
bool rsUp  = isRsi and xRsiUp
bool rsDn  = isRsi and xRsiDn

plotshape(sqUp or t3Up or rsUp, "Bullish trigger", shape.triangleup, location.bottom, cUp, size = size.tiny)
plotshape(sqDn or t3Dn or rsDn, "Bearish trigger", shape.triangledown, location.top, cDn, size = size.tiny)

// ----- Alerts
string msg = sqUp ? "squeeze fired UP" : sqDn ? "squeeze fired DOWN" : t3Up ? "3/10 crossed above signal" : t3Dn ? "3/10 crossed below signal" : rsUp ? "RSI back above " + str.tostring(rsiLo) : rsDn ? "RSI back below " + str.tostring(rsiHi) : ""
if msg != ""
    alert(syminfo.ticker + " " + timeframe.period + ": " + msg + " - last " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)

alertcondition(sqUp, "Squeeze fired up", "{{ticker}} {{interval}}: squeeze fired UP at {{close}}")
alertcondition(sqDn, "Squeeze fired down", "{{ticker}} {{interval}}: squeeze fired DOWN at {{close}}")
alertcondition(t3Up, "3/10 bullish cross", "{{ticker}} {{interval}}: 3/10 oscillator crossed above its signal")
alertcondition(t3Dn, "3/10 bearish cross", "{{ticker}} {{interval}}: 3/10 oscillator crossed below its signal")
alertcondition(rsUp, "RSI leaves oversold", "{{ticker}} {{interval}}: RSI crossed back above oversold")
alertcondition(rsDn, "RSI leaves overbought", "{{ticker}} {{interval}}: RSI crossed back below overbought")
DT Volume · volume bars that light up on spikes

Replaces the built-in Volume. Bars at 2× the recent average print in full color.

DT_Volume.pine · 33 lines

//@version=6
// ======================================================================
//  DT Volume - volume columns that light up on spikes
//  Replaces the built-in Volume indicator. Faded bars are ordinary volume;
//  full-color bars traded at least Nx the recent average (institutional
//  participation; Aziz and Bellafiore look for it on breakouts).
//  Educational tool. Not financial advice.
// ======================================================================
indicator("DT Volume", format = format.volume)

int   avgLen = input.int(20, "Average length", minval = 1)
float spikeX = input.float(2.0, "Spike threshold (x average)", minval = 1, step = 0.1)
color cUp    = input.color(#26a69a, "Up", inline = "c")
color cDn    = input.color(#ef5350, "Down", inline = "c")
color cAvg   = input.color(#f5a623, "Average", inline = "c")

float v     = nz(volume)
float vAvg  = ta.sma(v, avgLen)
bool  up    = close >= open
bool  spike = not na(vAvg) and vAvg > 0 and v >= spikeX * vAvg
color col   = spike ? (up ? cUp : cDn) : color.new(up ? cUp : cDn, 65)

plot(v, "Volume", col, style = plot.style_columns)
plot(vAvg, "Average", cAvg, 1)
plot(vAvg * spikeX, "Spike threshold", color.new(cAvg, 70), 1, display = display.none)

bool spikeUp = spike and up
bool spikeDn = spike and not up
if spike
    alert(syminfo.ticker + " " + timeframe.period + ": volume spike " + str.tostring(v / vAvg, "0.0") + "x average on a " + (up ? "green" : "red") + " bar - last " + str.tostring(close, format.mintick), alert.freq_once_per_bar_close)

alertcondition(spikeUp, "Volume spike on up bar", "{{ticker}} {{interval}}: volume spike on an up bar at {{close}}")
alertcondition(spikeDn, "Volume spike on down bar", "{{ticker}} {{interval}}: volume spike on a down bar at {{close}}")
